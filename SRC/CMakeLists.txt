
file(GLOB sources "*.c")

function(add_superlu_mt_target PLAT_SUFFIX)
  add_library(superlu_mt${PLAT_SUFFIX} ${sources})
  set_target_properties(superlu_mt${PLAT_SUFFIX} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  set_property(TARGET superlu_mt${PLAT_SUFFIX} PROPERTY C_STANDARD 99)
  if(HAVE_LIB_M)
    target_link_libraries(superlu_mt${PLAT_SUFFIX} PRIVATE m)
  endif()
  target_link_libraries(superlu_mt${PLAT_SUFFIX} PRIVATE ${BLAS_LIB})
  target_compile_definitions(superlu_mt${PLAT_SUFFIX} PRIVATE Add_)
  if(PLAT_SUFFIX STREQUAL "_PTHREAD")
    target_compile_definitions(superlu_mt${PLAT_SUFFIX} PUBLIC __PTHREAD)
    target_link_libraries(superlu_mt${PLAT_SUFFIX} PRIVATE ${CMAKE_THREAD_LIBS_INIT})
  elseif(PLAT_SUFFIX STREQUAL "_OPENMP")
    target_compile_definitions(superlu_mt${PLAT_SUFFIX} PUBLIC __OPENMP)
    target_link_libraries(superlu_mt${PLAT_SUFFIX} PRIVATE OpenMP::OpenMP_C)
  endif()
  if(LONGINT)
    target_compile_definitions(superlu_mt${PLAT_SUFFIX} PUBLIC _LONGINT)
  endif()
  target_include_directories(superlu_mt${PLAT_SUFFIX} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_include_directories(superlu_mt${PLAT_SUFFIX} INTERFACE $<INSTALL_INTERFACE:include>)
  install(TARGETS superlu_mt${PLAT_SUFFIX} DESTINATION ${CMAKE_INSTALL_LIBDIR})
endfunction()

if(THREAD_API STREQUAL "PTHREAD" OR THREAD_API STREQUAL "BOTH")
  add_superlu_mt_target(_PTHREAD)
endif()
if(THREAD_API STREQUAL "OPENMP" OR THREAD_API STREQUAL "BOTH")
  add_superlu_mt_target(_OPENMP)
endif()

file(GLOB headers "*.h")
install(FILES ${headers} DESTINATION ${SUPERLUMT_INSTALL_INCLUDEDIR})
